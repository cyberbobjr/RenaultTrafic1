/* Renault Trafic I (T800, short wheelbase, low roof, blind panel van).
   Values are written in real metres and multiplied by the model scale (0.85) at load
   (VehicleScript.Loaded): 3.69 x 1.62 x 1.67 m in game, same uniform reduction for every axis.
   Vehicle frame: x > 0 = left, z > 0 = front. The FBX origin remains 1.0 m above the ground.
   The physics origin is lowered by moving the model and collision +0.40 m in that frame;
   spawn height compensates -0.40 m. Meshes, animations, wheels and passenger positions retain
   their model-space coordinates. The script centre offset keeps the Java bounding box aligned;
   it is not a separate native centre-of-mass parameter when custom physics boxes are used.
   Base: vanilla Van (templates, sounds, mechanics overlay). First pass: offsets to be checked
   in game with the debug vehicle editor. */

        extents = 1.90 1.60 4.34,
        /* Low physics-origin trial: translate model offset, collision offset and Java bounding
           centre +0.40 real metres together (0.34 m in game). CarController compensates the
           initial height using wheel.offset + model.offset - radius. toBullet includes model.offset
           in wheel anchors but NOT in custom physics boxes, hence the matching +0.40 below.
           Static world-space positions are preserved; the intended reduction in pitching relies
           on the native rigid-body origin and must be checked in game.
           centerOfMassOffset is only used for Java bounds and spawn here: toBullet does not send
           it as a separate centre-of-mass value with useChassisPhysicsCollision=false and a box.
           The box bottom stays 0.50 m above the ground in real dimensions. No VanCollision. */
        useChassisPhysicsCollision = false,
        physicsChassisShape = 1.90 1.50 4.34,
        centerOfMassOffset = 0.0 0.40 0.0,
        physics box

            offset = 0.0 0.65 0.0,

            rotate = 0.0 0.0 0.0,

        mass = 1250,
        engineForce = 3500,
        engineQuality = 70,
        engineLoudness = 90,
        maxSpeed = 60.0,
        brakingForce = 70,
        stoppingMovementForce = 5.0,
        /* rollInfluence limits lateral-force roll; the low-origin trial also targets pitch.
           The centre-of-mass position inside the native PZ solver is not confirmed by Java. */
        rollInfluence = 0.45,
        steeringIncrement = 0.03,
        steeringClamp = 0.3,
        /* Trial after confirmed pitching with stiffness 90: firmer springs, stronger damping.
           These damping values are healthy type-2 suspension targets, applied per part by
           client/batman_RenaultTraficI_Physics.lua. BaseVehicle.updatePartStats replaces part
           damping with inventory-item values; updateBulletStatsWheel sends those to Bullet,
           overwriting the script's initial damping. Script-only changes were insufficient.
           The calibration preserves item condition and leaves missing suspensions missing.
           Travel scales from 16 to 13.6 cm; stiffness and damping are not scaled.
           Collision geometry and wheel anchors stay as defined above. In-game trial required. */
        suspensionStiffness = 120.0,
        suspensionCompression = 12.0,
        suspensionDamping = 10.0,
        maxSuspensionTravelCm = 16.0,
        /* 0.40 (0.34 m in game): the physics hangs each wheel from offset.y + suspensionRestLength
           (VehicleScript, Bullet.defineVehicleScript) and casts its ray down by rest length + radius.
           With 0.2 the anchor sat just above the chassis box bottom and 0.3 m above the ground: hard
           pitching (fast reverse then forward) sank the van; probable cause: the anchors went below
           the ground and the rays missed. Vanilla Van ~0.36 m, KI5 Type 2 0.27 m. The drawn wheel stays at offset.y at
           rest (BaseVehicle.updateTransform), so the look does not change. */
        suspensionRestLength = 0.40,
