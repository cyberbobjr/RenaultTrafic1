/* Renault Trafic I - models.
   One FBX (common/media/models_X/vehicles/Vehicles_batman_RenaultTraficI.fbx) holds every part as
   a named node, the skeleton (trafic_root, door_fl/fr_bone, window_fl/fr_bone, door_rr_bone,
   trunk_l/r_bone, hood_bone) and the *_opening animations, laid out like the vanilla
   ModernCarWithDoors.blend (Assets/tools/trafic_export_fbx.py). The skeleton node is named
   VehicleSkeleton (the name jassimp/ImportedSkeleton looks for) and every FBX node has an identity
   transform: vertices and bones already hold the game coordinates in metres, hence scale 1.0.
   Body/opaque parts use vanilla vehicle_multiuv_noreflect: the game appends "_static"
   for static models (core/opengl/ShaderProgram) and vehicle_multiuv has no animated vertex
   program, so the moving parts cannot use it; and vehicle_multiuv.frag blends the body towards a
   quarter of the sky reflection (up to 35 %), which made the body darker than the doors. Same
   shader on opaque parts = same shade everywhere. The three glass models use the authorised
   standalone batman_trafic_glass (same vehicle interface, alpha 0.28 on glass, opaque seals;
   no damnlib). Wheels use vehiclewheel and scale 1.12.
   Each animated part is masked to its own bone (boneWeight, without descendants). A child part
   (window, parent = DoorFrontLeft) shares its parent's animation player
   (BaseVehicle.ModelInfo.getAnimationPlayer): the door track moves the door bone, the window
   track moves the window bone, so the window follows its door. Masked bones keep an identity
   local transform, hence VehicleSkeleton and trafic_root rest at the identity in the FBX. */
module Base
{
    model batman_TraficI_Body
    {
        mesh = vehicles/Vehicles_batman_RenaultTraficI|TrafI_body,
        shader = vehicle_multiuv_noreflect,
        invertX = false,
        static = true,
        scale = 1.0,
    }

    model batman_TraficI_Interior
    {
        mesh = vehicles/Vehicles_batman_RenaultTraficI|TrafI_interior,

    }
